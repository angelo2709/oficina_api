CREATE DATABASE oficina;
USE oficina;

-- ============================================================
-- CATEGORIAS
-- ============================================================

CREATE TABLE categorias (
    categoria_id INT AUTO_INCREMENT PRIMARY KEY,
    categoria_nome VARCHAR(50) NOT NULL UNIQUE,
    categoria_descricao VARCHAR(150)
);


-- ============================================================
-- FABRICANTES
-- ============================================================

CREATE TABLE fabricantes (
    fabricante_id INT AUTO_INCREMENT PRIMARY KEY,
    fabricante_nome VARCHAR(50) NOT NULL UNIQUE,
    fabricante_cnpj VARCHAR(20) UNIQUE,
    fabricante_pais VARCHAR(50)
);


-- ============================================================
-- PECAS
-- ============================================================

CREATE TABLE pecas (
    peca_id INT AUTO_INCREMENT PRIMARY KEY,
    categoria_id INT NOT NULL,
    fabricante_id INT NOT NULL,
    peca_sku VARCHAR(30) NOT NULL UNIQUE,
    peca_nome VARCHAR(100) NOT NULL,
    peca_preco_venda DECIMAL(10,2) NOT NULL,
    peca_estoque_atual INT NOT NULL DEFAULT 0,

    FOREIGN KEY (categoria_id)
        REFERENCES categorias(categoria_id)
        ON DELETE RESTRICT,

    FOREIGN KEY (fabricante_id)
        REFERENCES fabricantes(fabricante_id)
        ON DELETE RESTRICT
);


-- ============================================================
-- FORNECEDORES
-- ============================================================

CREATE TABLE fornecedores (
    fornecedor_id INT AUTO_INCREMENT PRIMARY KEY,
    fornecedor_razao_social VARCHAR(100) NOT NULL,
    fornecedor_cnpj VARCHAR(20) NOT NULL UNIQUE,
    fornecedor_telefone VARCHAR(20),
    fornecedor_email VARCHAR(100)
);


-- ============================================================
-- FORNECEDORES X PECAS
-- RELACIONAMENTO M:N
-- ============================================================

CREATE TABLE fornecedores_pecas (
    fornecedor_peca_id INT AUTO_INCREMENT PRIMARY KEY,
    fornecedor_id INT NOT NULL,
    peca_id INT NOT NULL,
    fornecedor_peca_preco_custo DECIMAL(10,2) NOT NULL,
    fornecedor_peca_prazo_entrega_dias INT DEFAULT 1,
    fornecedor_peca_codigo VARCHAR(50),

    FOREIGN KEY (fornecedor_id)
        REFERENCES fornecedores(fornecedor_id)
        ON DELETE RESTRICT,

    FOREIGN KEY (peca_id)
        REFERENCES pecas(peca_id)
        ON DELETE RESTRICT,

    UNIQUE (
        fornecedor_id,
        peca_id
    )
);

-- ============================================================
-- CONSULTA 1
-- LISTAR PECAS COM CATEGORIA E FABRICANTE
-- ============================================================

SELECT
    p.peca_id,
    p.peca_nome,
    p.peca_sku,
    c.categoria_nome AS categoria,
    f.fabricante_nome AS fabricante,
    p.peca_preco_venda,
    p.peca_estoque_atual

FROM pecas p

JOIN categorias c
    ON p.categoria_id = c.categoria_id

JOIN fabricantes f
    ON p.fabricante_id = f.fabricante_id;


-- ============================================================
-- CONSULTA 2
-- FORNECEDORES DE CADA PECA
-- ============================================================

SELECT
    p.peca_nome AS peca,
    f.fornecedor_razao_social AS fornecedor,
    fp.fornecedor_peca_preco_custo AS preco_custo,
    fp.fornecedor_peca_prazo_entrega_dias AS prazo_entrega

FROM fornecedores_pecas fp

JOIN fornecedores f
    ON fp.fornecedor_id = f.fornecedor_id

JOIN pecas p
    ON fp.peca_id = p.peca_id

ORDER BY
    p.peca_nome,
    fp.fornecedor_peca_preco_custo;


-- ============================================================
-- CONSULTA 3
-- PECAS FORNECIDAS POR UM DETERMINADO FORNECEDOR
-- ============================================================

SELECT
    f.fornecedor_razao_social AS fornecedor,
    p.peca_nome AS peca,
    fp.fornecedor_peca_preco_custo AS preco_custo

FROM fornecedores_pecas fp

JOIN fornecedores f
    ON fp.fornecedor_id = f.fornecedor_id

JOIN pecas p
    ON fp.peca_id = p.peca_id

WHERE f.fornecedor_id = 1;


-- ============================================================
-- CONSULTA 4
-- COMPARAR PRECO DE CUSTO E PRECO DE VENDA
-- ============================================================

SELECT
    p.peca_nome AS peca,
    f.fornecedor_razao_social AS fornecedor,
    fp.fornecedor_peca_preco_custo AS preco_custo,
    p.peca_preco_venda AS preco_venda,

    (
        p.peca_preco_venda -
        fp.fornecedor_peca_preco_custo
    ) AS diferenca

FROM fornecedores_pecas fp

JOIN fornecedores f
    ON fp.fornecedor_id = f.fornecedor_id

JOIN pecas p
    ON fp.peca_id = p.peca_id

ORDER BY
    p.peca_nome,
    fp.fornecedor_peca_preco_custo;
