<section>
    <h2>4. Configurar a singularização em português</h2>
    <p>O scaffolding gera nomes errados (ex: <code class="inline">FornecedoresPecas.cs</code>). Vamos corrigir isso.</p>
    
    <h3>4.1 Criar a pasta Design e o arquivo PortuguesePluralizer.cs</h3>
    <div class="code-wrap">
        <button class="copy">Copiar</button>
        <pre>using Microsoft.EntityFrameworkCore.Design;

namespace oficina_api.Design;

public class PortuguesePluralizer : IPluralizer
{
    public string Singularize(string identifier)
    {
        return identifier.ToLower() switch
        {
            "categorias" => "Categoria",
            "fabricantes" => "Fabricante",
            "pecas" => "Peca",
            "fornecedores" => "Fornecedor",
            "fornecedorespecas" => "FornecedorPeca",
            "fornecedores_pecas" => "FornecedorPeca",
            _ => identifier
        };
    }

    public string Pluralize(string identifier)
    {
        return identifier switch
        {
            "Categoria" => "Categorias",
            "Fabricante" => "Fabricantes",
            "Peca" => "Pecas",
            "Fornecedor" => "Fornecedores",
            "FornecedorPeca" => "FornecedoresPecas",
            _ => identifier
        };
    }
}</pre>
    </div>

    <h3>4.2 Criar o arquivo DesignTimeServices.cs</h3>
    <div class="code-wrap">
        <button class="copy">Copiar</button>
        <pre>using Microsoft.EntityFrameworkCore.Design;
using Microsoft.Extensions.DependencyInjection;

namespace oficina_api.Design;

public class DesignTimeServices : IDesignTimeServices
{
    public void ConfigureDesignTimeServices(IServiceCollection services)
    {
        services.AddSingleton&lt;IPluralizer, PortuguesePluralizer&gt;();
    }
}</pre>
    </div>

    <h3>4.3 Ajustar o .csproj</h3>
    <p>No arquivo <code class="inline">oficina_api.csproj</code>, garanta que o pacote Design esteja assim:</p>
    <div class="code-wrap">
        <button class="copy">Copiar</button>
        <pre>&lt;PackageReference Include="Microsoft.EntityFrameworkCore.Design" Version="10.0.12"&gt;
  &lt;PrivateAssets&gt;all&lt;/PrivateAssets&gt;
&lt;/PackageReference&gt;</pre>
    </div>
    <p>Rode <code class="inline">dotnet build</code> antes de prosseguir.</p>
</section>

<section>
    <h2>6. Configurar o Program.cs e o Banco</h2>

    <h3>6.1 appsettings.json</h3>
    <div class="code-wrap">
        <button class="copy">Copiar</button>
        <pre>{
  "ConnectionStrings": {
    "DefaultConnection": "server=localhost;database=oficina;user=root;password=;"
  },
  "Logging": {
    "LogLevel": {
      "Default": "Information",
      "Microsoft.AspNetCore": "Warning"
    }
  },
  "AllowedHosts": "*"
}</pre>
    </div>

    <h3>6.2 Program.cs (Com correção de ciclos e HTTPS)</h3>
    <div class="code-wrap">
        <button class="copy">Copiar</button>
        <pre>using oficina_api.Data;
using Microsoft.EntityFrameworkCore;
using System.Text.Json.Serialization;

var builder = WebApplication.CreateBuilder(args);

// Configuração de Portas (Substitui o launchSettings.json ausente)
builder.WebHost.UseUrls("https://localhost:7000;http://localhost:7001");

// Adiciona Controllers e Configura o Serializador para ignorar ciclos
builder.Services.AddControllers()
    .AddJsonOptions(options =>
    {
        options.JsonSerializerOptions.ReferenceHandler = ReferenceHandler.IgnoreCycles;
    });

builder.Services.AddDbContext&lt;OficinaContext&gt;(options =>
{
    var connectionString = builder.Configuration.GetConnectionString("DefaultConnection");
    options.UseMySQL(connectionString!);
});

builder.Services.AddOpenApi();

var app = builder.Build();

if (app.Environment.IsDevelopment())
{
    app.MapOpenApi();
}

app.UseHttpsRedirection();
app.UseAuthorization();
app.MapControllers();
app.Run();</pre>
    </div>
